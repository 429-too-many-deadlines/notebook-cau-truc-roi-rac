\begin{frame}{Mệnh Đề là Gì? (Bản chất Nhị phân)}
  \begin{cozyDef}[Định nghĩa Mệnh đề (Proposition)]
    Một \textbf{mệnh đề} là một câu khẳng định một sự kiện khách quan, có giá trị chân lý hoặc \textbf{Đúng} ($1$) hoặc \textbf{Sai} ($0$), tuyệt đối không thể vừa đúng vừa sai hoặc không xác định.
  \end{cozyDef}
  \vspace{0.05cm}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{cozyInsight}[Ví dụ về Mệnh Đề]
        \small
        \begin{itemize}
          \setlength{\itemsep}{1pt}
          \item ``Hà Nội là thủ đô của Việt Nam.''\\
          $\to$ \textbf{\textcolor{cozyTrue}{Đúng (1)}}
          \item ``$0 > 2$''\\
          $\to$ \textbf{\textcolor{cozyFalse}{Sai (0)}}
          \item ``$3$ là một số nguyên tố.''\\
          $\to$ \textbf{\textcolor{cozyTrue}{Đúng (1)}}
        \end{itemize}
      \end{cozyInsight}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{cozyTrap}[KHÔNG Phải Mệnh Đề]
        \small
        \begin{itemize}
          \setlength{\itemsep}{1pt}
          \item ``Một ngày đẹp trời làm sao!''\\
          \textcolor{cozyMuted}{(Câu cảm thán)}
          \item ``Thầy Hợi rất đẹp trai!''\\
          \textcolor{cozyMuted}{(Đánh giá chủ quan)}
          \item ``$3 + x = 2$''\\
          \textcolor{cozyMuted}{(Vị từ chứa biến, chưa biết $x$)}
          \item ``Câu này là sai.''\\
          \textcolor{cozyMuted}{(Nghịch lý tự quy chiếu)}
        \end{itemize}
      \end{cozyTrap}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ký Hiệu, Chân Trị \& Góc Nhìn Khoa Học Máy Tính}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{cozyDef}[Ký hiệu \& Chân trị (Truth Value)]
        \small
        \begin{itemize}
          \item \textbf{Ký hiệu:} $p, q, r, s, \dots$
          \item \textbf{Chân trị:} Mỗi mệnh đề nhận duy nhất một trong hai giá trị:
          \begin{itemize}
            \item \textbf{Đúng} (True): Ký hiệu \textbf{Đ} hoặc $\mathbf{1}$.
            \item \textbf{Sai} (False): Ký hiệu \textbf{S} hoặc $\mathbf{0}$.
          \end{itemize}
          \item Tuyệt đối không có trạng thái thứ ba.
        \end{itemize}
      \end{cozyDef}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{cozyInsight}[Góc nhìn Khoa học Máy tính]
        \small
        \begin{itemize}
          \item Trong máy tính, chân lý nhị phân chính là \textbf{Bit ($1 / 0$)}, là mức điện thế Cao/Thấp trong vi mạch bán dẫn.
          \item Trong lập trình, đây chính là kiểu dữ liệu nguyên thủy \textbf{\texttt{bool}} (\texttt{True} / \texttt{False}), quyết định mọi luồng rẽ nhánh \texttt{if / else}, vòng lặp và logic điều khiển trong phần mềm.
        \end{itemize}
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phân Loại Mệnh Đề: Sơ Cấp vs Phức Hợp}
  \begin{cozyDef}[1. Mệnh đề Sơ cấp (Nguyên tử -- Atomic Proposition)]
    \small
    Chỉ chứa duy nhất một sự kiện, không thể chia nhỏ hơn thành các mệnh đề khác.
    \begin{itemize}
      \item \textit{Ví dụ:} ``$3$ là số nguyên tố'', ``$2$ là số chẵn''.
    \end{itemize}
  \end{cozyDef}
  \vspace{0.15cm}
  \begin{cozyMethod}[2. Mệnh đề Phức hợp (Compound Proposition)]
    \small
    Được hình thành bằng cách kết hợp các mệnh đề sơ cấp lại với nhau thông qua các \textbf{toán tử logic} (liên từ: \textit{và, hoặc, phủ định, nếu\dots thì, nếu và chỉ nếu}).
    \begin{itemize}
      \item \textit{Ví dụ 1:} ``$2$ là số nguyên tố \textbf{\color{cozyWarm}VÀ} là một số chẵn.''
      \item \textit{Ví dụ 2:} ``Thầy Hợi đang dạy học \textbf{\color{cozyWarm}HOẶC} đang xem phim.''
    \end{itemize}
    \vspace{0.1cm}
    \par
    \textbf{\color{cozySlate}Lưu ý sư phạm:} Trong toán học, mệnh đề phức hợp được mô hình hóa như một hàm chân trị nhận các biến mệnh đề đầu vào.
  \end{cozyMethod}
\end{frame}

\begin{frame}{1. Phép Phủ Định (NOT -- $\neg p$ hoặc $\bar{p}$)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{cozyDef}[Định nghĩa \& Bảng Chân Lý]
        \small
        Phủ định của mệnh đề $p$ là mệnh đề \textbf{``Không phải trường hợp $p$''}, ký hiệu là $\neg p$ hoặc $\bar{p}$.\\
        \vspace{0.05cm}
        Giá trị chân lý của $\bar{p}$ luôn \textbf{ngược lại} với $p$: đúng khi $p$ sai, và sai khi $p$ đúng.
        \vspace{0.2cm}
        \begin{center}
          \renewcommand{\arraystretch}{1.15}
          \begin{tabular}{cc}
            \toprule
            \textbf{Mệnh đề ($p$)} & \textbf{Phủ định ($\bar{p}$)} \\
            \midrule
            $0$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $1$ & \textbf{\textcolor{cozyFalse}{0}} \\
            \bottomrule
          \end{tabular}
        \end{center}
        \vspace{0.1cm}
        \par\footnotesize\centering\color{cozyMuted}
        Đảo bit: $0 \leftrightarrow 1$ (Cổng NOT logic)
      \end{cozyDef}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{cozyInsight}[Ví dụ Minh Họa từ Thực Tế]
        \small
        \begin{itemize}
          \setlength{\itemsep}{4pt}
          \item $p$: ``Paris là thủ đô của Pháp.'' (\textbf{\textcolor{cozyTrue}{1}})\\
          $\implies \bar{p}$: ``Paris \textbf{không phải} là thủ đô của Pháp.'' (\textbf{\textcolor{cozyFalse}{0}})
          \item $q$: ``$9$ là một số chính phương.'' (\textbf{\textcolor{cozyTrue}{1}})\\
          $\implies \bar{q}$: ``$9$ \textbf{không phải} là một số chính phương.'' (\textbf{\textcolor{cozyFalse}{0}})
          \item $r$: ``$0 > 2$'' (\textbf{\textcolor{cozyFalse}{0}})\\
          $\implies \bar{r}$: ``$0 \le 2$'' (\textbf{\textcolor{cozyTrue}{1}})
        \end{itemize}
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{2. Phép Tuyển Thường (OR -- $p \lor q$)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{cozyDef}[Định nghĩa \& Bảng Chân Lý]
        \small
        Phép tuyển của $p$ và $q$ là mệnh đề \textbf{``$p$ hoặc $q$''}, ký hiệu là $p \lor q$.\\
        \vspace{0.05cm}
        $p \lor q$ \textbf{\textcolor{cozyTrue}{Đúng}} khi có \textbf{ít nhất một} mệnh đề thành phần đúng; chỉ \textbf{\textcolor{cozyFalse}{Sai}} khi cả $p$ và $q$ đều sai.
        \vspace{0.15cm}
        \begin{center}
          \renewcommand{\arraystretch}{1.1}
          \begin{tabular}{cc|c}
            \toprule
            $p$ & $q$ & $p \lor q$ \\
            \midrule
            $0$ & $0$ & \textbf{\textcolor{cozyFalse}{0}} \\
            $0$ & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $1$ & $0$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $1$ & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
            \bottomrule
          \end{tabular}
        \end{center}
        \vspace{0.05cm}
        \par\footnotesize\centering\color{cozyMuted}Chỉ cần 1 đầu vào đúng $\implies$ Đúng
      \end{cozyDef}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{cozyInsight}[Bản chất Inclusive OR \& Ví dụ]
        \small
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item \textbf{Inclusive OR (Tuyển bao hàm):} Trong toán học, ``hoặc'' cho phép \textbf{cả hai cùng xảy ra}.
          \item \textit{Ví dụ xét học bổng:} ``Sinh viên có GPA $\ge 8.0$ \textbf{HOẶC} có bài báo nghiên cứu khoa học.''\\
          $\to$ Đạt \textbf{cả hai} điều kiện thì càng xứng đáng!
          \item \textit{Ví dụ slide gốc:} ``$3$ là số nguyên tố \textbf{hoặc} $3$ là số chẵn.''\\
          $\to$ Vì $3$ là số nguyên tố (Đúng) $\implies$ Toàn bộ mệnh đề \textbf{\textcolor{cozyTrue}{Đúng (1)}}.
        \end{itemize}
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{3. Phép Hội (AND -- $p \land q$)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{cozyDef}[Định nghĩa \& Bảng Chân Lý]
        \small
        Phép hội của $p$ và $q$ là mệnh đề \textbf{``$p$ và $q$''}, ký hiệu là $p \land q$.\\
        \vspace{0.05cm}
        $p \land q$ chỉ \textbf{\textcolor{cozyTrue}{Đúng}} khi \textbf{cả hai} mệnh đề đều đúng; chỉ cần một mệnh đề sai thì kết quả sẽ \textbf{\textcolor{cozyFalse}{Sai}}.
        \vspace{0.15cm}
        \begin{center}
          \renewcommand{\arraystretch}{1.1}
          \begin{tabular}{cc|c}
            \toprule
            $p$ & $q$ & $p \land q$ \\
            \midrule
            $0$ & $0$ & \textbf{\textcolor{cozyFalse}{0}} \\
            $0$ & $1$ & \textbf{\textcolor{cozyFalse}{0}} \\
            $1$ & $0$ & \textbf{\textcolor{cozyFalse}{0}} \\
            $1$ & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
            \bottomrule
          \end{tabular}
        \end{center}
        \vspace{0.05cm}
        \par\footnotesize\centering\color{cozyMuted}Cả hai cùng đúng $\implies$ Mới đúng
      \end{cozyDef}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{cozyMethod}[Đối Chiếu Trực Quan: $\lor$ (OR) vs $\land$ (AND)]
        \small
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item \textbf{Phép $\lor$ rất ``dễ tính'':} Chỉ cần ít nhất $1$ nhánh đúng là đúng (mạch điện \textit{song song}).
          \item \textbf{Phép $\land$ rất ``khắt khe'':} Bắt buộc cả hai nhánh phải đúng (mạch điện \textit{nối tiếp}).
          \item \textit{Ví dụ slide gốc:} ``Trời đã mưa \textbf{VÀ} hôm đó là thứ Sáu.''\\
          $\to$ Chỉ đúng khi đồng thời cả 2 sự kiện xảy ra; sai 1 sự kiện $\implies$ phát biểu sai.
        \end{itemize}
      \end{cozyMethod}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{4. Phép Tuyển Loại (XOR -- $p \oplus q$)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{cozyDef}[Định nghĩa \& Bảng Chân Lý]
        \small
        Phép tuyển loại của $p$ và $q$ là mệnh đề \textbf{``$p$ hoặc $q$ nhưng không cả hai''}, ký hiệu $p \oplus q$.\\
        \vspace{0.05cm}
        $p \oplus q$ \textbf{\textcolor{cozyTrue}{Đúng}} khi $p, q$ có \textbf{chân trị khác nhau} (chỉ đúng $1$ trong $2$); \textbf{\textcolor{cozyFalse}{Sai}} khi cùng chân trị.
        \vspace{0.15cm}
        \begin{center}
          \renewcommand{\arraystretch}{1.1}
          \begin{tabular}{cc|c}
            \toprule
            $p$ & $q$ & $p \oplus q$ \\
            \midrule
            $0$ & $0$ & \textbf{\textcolor{cozyFalse}{0}} \\
            $0$ & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $1$ & $0$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $1$ & $1$ & \textbf{\textcolor{cozyFalse}{0}} \;\textcolor{cozyWarm}{$\leftarrow$ Khác $\lor$} \\
            \bottomrule
          \end{tabular}
        \end{center}
        \vspace{0.05cm}
        \par\footnotesize\centering\color{cozyMuted}Cộng nhị phân không nhớ: $1 \oplus 1 = 0$
      \end{cozyDef}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{cozyTrap}[Phân biệt Cốt lõi: Tuyển thường $\lor$ vs Tuyển loại $\oplus$]
        \small
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item \textbf{$\lor$ (Inclusive OR):} Cho phép chọn cả hai. Ví dụ: ``Biết tiếng Anh hoặc tiếng Nhật''.
          \item \textbf{$\oplus$ (Exclusive OR):} Bắt buộc chỉ chọn duy nhất 1 trong 2:
          \begin{itemize}
            \setlength{\itemsep}{2pt}
            \item \textit{Thực đơn cơm trưa:} ``Chọn canh chua HOẶC canh rau'' (chỉ được chọn đúng 1 món).
            \item \textit{Ví dụ slide gốc:} ``Chọn súp miễn phí hoặc salad miễn phí kèm bữa ăn.'' Nếu cố lấy cả hai $\implies$ Phạm quy (\textbf{\textcolor{cozyFalse}{Sai}})!
          \end{itemize}
        \end{itemize}
      \end{cozyTrap}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{5. Phép Kéo Theo (Conditional -- $p \to q$)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{cozyDef}[Định nghĩa \& Bảng Chân Lý]
        \small
        Phép kéo theo của $p$ và $q$ là mệnh đề \textbf{``Nếu $p$ thì $q$''}, ký hiệu là $p \to q$.\\
        \vspace{0.05cm}
        Trong đó: $p$ là \textbf{Tiền đề} (Giả thiết), $q$ là \textbf{Kết luận} (Hệ quả).
        \vspace{0.15cm}
        \begin{center}
          \renewcommand{\arraystretch}{1.1}
          \begin{tabular}{cc|c}
            \toprule
            $p$ & $q$ & $p \to q$ \\
            \midrule
            $0$ & $0$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $0$ & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $1$ & $0$ & \textbf{\textcolor{cozyFalse}{0}} \;\textcolor{cozyFalse}{$\leftarrow$ DUY NHẤT SAI} \\
            $1$ & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
            \bottomrule
          \end{tabular}
        \end{center}
        \vspace{0.05cm}
        \par\footnotesize\centering\color{cozyMuted}Tiền đề sai $\implies$ Luôn luôn Đúng
      \end{cozyDef}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{cozyTrap}[Quy Tắc Vàng Ghi Nhớ $p \to q$]
        \small
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item Mệnh đề kéo theo $p \to q$ chỉ \textbf{\textcolor{cozyFalse}{Sai}} trong \textbf{duy nhất một trường hợp}:\\
          \vspace{0.1cm}
          \centerline{\textbf{\textcolor{cozySlate}{Tiền đề Đúng ($p=1$)}} $\implies$ \textbf{\textcolor{cozyFalse}{Kết luận Sai ($q=0$)}}}
          \vspace{0.1cm}
          \item Trong 3 trường hợp còn lại, $p \to q$ luôn \textbf{\textcolor{cozyTrue}{Đúng ($1$)}}.
          \item \textit{Ví dụ slide gốc:} ``Nếu bạn học chăm chỉ ($p$), thì bạn sẽ qua môn ($q$).''
        \end{itemize}
      \end{cozyTrap}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Giải Mã Phép Kéo Theo: Vì sao Tiền đề Sai thì Luôn Đúng?}
  \begin{cozyInsight}[Mô hình Lời Hứa / Hợp Đồng Thưởng Điểm]
    \small
    Thầy giáo hứa với học sinh: \textit{``Nếu em đạt điểm 10 môn Toán ($p$), thầy sẽ tặng em máy tính ($q$).''}\\
    Trong trường hợp nào thì thầy giáo bị coi là \textbf{nói dối / thất hứa} ($p \to q$ có giá trị Sai)?
    \begin{itemize}
      \setlength{\itemsep}{2pt}
      \item \textbf{Trường hợp 1 ($p=1, q=1$):} Em được 10 điểm, thầy tặng máy $\implies$ Thầy \textbf{giữ lời} (\textbf{\textcolor{cozyTrue}{Đúng}}).
      \item \textbf{Trường hợp 2 ($p=1, q=0$):} Em được 10 điểm, thầy không tặng $\implies$ Thầy \textbf{thất hứa} (\textbf{\textcolor{cozyFalse}{Sai}}).
      \item \textbf{Trường hợp 3 \& 4 ($p=0$):} Em \textbf{KHÔNG} được 10 điểm! Lúc này dù thầy có tặng hay không tặng máy tính, thầy đều \textbf{KHÔNG THẤT HỨA} $\implies$ Lời hứa hiển nhiên \textbf{\textcolor{cozyTrue}{Đúng}}!
    \end{itemize}
  \end{cozyInsight}
  \vspace{0.05cm}
  \begin{cozyDef}[Ghi Nhớ Cốt Lõi: Chân Trị Chân Không (Vacuous Truth)]
    \small
    Khi tiền đề $p$ \textbf{Sai ($0$)}, mệnh đề kéo theo $p \to q$ tự động nhận giá trị \textbf{\textcolor{cozyTrue}{Đúng ($1$)}} bất kể $q$ Đúng hay Sai. Trong logic học, hiện tượng này được gọi là \textbf{Vacuous Truth} (Đúng chân không).
  \end{cozyDef}
\end{frame}

\begin{frame}{Cạm Bẫy Thuật Ngữ: Điều Kiện Cần vs Điều Kiện Đủ}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{cozyTrap}[Phân biệt Điều kiện Cần và Đủ]
        \small
        Trong mệnh đề kéo theo $p \to q$:
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item \textbf{$p$ là điều kiện ĐỦ để có $q$}:\\
          Chỉ cần có $p$ là chắc chắn có $q$. Có $p$ là ``đủ'' để suy ra $q$.
          \item \textbf{$q$ là điều kiện CẦN để có $p$}:\\
          Muốn có $p$ thì bắt buộc phải có $q$. Nếu không có $q$ thì chắc chắn không thể có $p$ ($\neg q \to \neg p$).
        \end{itemize}
      \end{cozyTrap}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{cozyMethod}[Công Thức Cốt Lõi: \mbox{$p \to q \equiv \neg p \lor q$}]
        \small
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item \textbf{Ý nghĩa trực giác:}\\
          ``Để không vi phạm lời hứa, hoặc là không có tiền đề ràng buộc ($\neg p$), hoặc là đã thực hiện kết luận ($q$).''
          \item \textbf{Ví dụ đời sống:}\\
          ``Trời mưa ($p$) $\to$ Đường ướt ($q$)''.
          \begin{itemize}
            \item Trời mưa là điều kiện \textit{đủ} để làm đường ướt.
            \item Đường ướt là điều kiện \textit{cần} để khẳng định trời đã mưa.
          \end{itemize}
        \end{itemize}
      \end{cozyMethod}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{6. Phép Tương Đương (Biconditional -- $p \leftrightarrow q$)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{cozyDef}[Định nghĩa \& Bảng Chân Lý]
        \small
        Mệnh đề tương đương của $p$ và $q$ là \textbf{``$p$ nếu và chỉ nếu $q$''}, ký hiệu $p \leftrightarrow q$.\\
        \vspace{0.05cm}
        $p \leftrightarrow q$ \textbf{\textcolor{cozyTrue}{Đúng}} khi $p, q$ có \textbf{cùng chân trị} ($1 \leftrightarrow 1$ là $1$, $0 \leftrightarrow 0$ là $1$); \textbf{\textcolor{cozyFalse}{Sai}} khi khác chân trị.
        \vspace{0.15cm}
        \begin{center}
          \renewcommand{\arraystretch}{1.1}
          \begin{tabular}{cc|c}
            \toprule
            $p$ & $q$ & $p \leftrightarrow q$ \\
            \midrule
            $0$ & $0$ & \textbf{\textcolor{cozyTrue}{1}} \\
            $0$ & $1$ & \textbf{\textcolor{cozyFalse}{0}} \\
            $1$ & $0$ & \textbf{\textcolor{cozyFalse}{0}} \\
            $1$ & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
            \bottomrule
          \end{tabular}
        \end{center}
        \vspace{0.05cm}
        \par\footnotesize\centering\color{cozyMuted}Cùng chân trị $\implies 1$; Khác chân trị $\implies 0$
      \end{cozyDef}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{cozyInsight}[Bản chất Kéo Theo 2 Chiều \& Ví dụ]
        \small
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item \textbf{Bản chất là kéo theo 2 chiều:}
          $$p \leftrightarrow q \equiv (p \to q) \land (q \to p)$$
          $p$ vừa là điều kiện cần, vừa là điều kiện đủ của $q$.
          \item \textit{Ví dụ từ giáo trình:}\\
          ``Bạn có thể lên chuyến bay nếu và chỉ nếu bạn mua vé.''\\
          {\footnotesize\color{cozyMuted}(với $p$: ``Lên chuyến bay'', $q$: ``Mua vé'')}
          \begin{itemize}
            \setlength{\itemsep}{2pt}
            \item Lên chuyến bay $\implies$ Đã mua vé ($p \to q$).
            \item Có vé $\implies$ Được lên chuyến bay ($q \to p$).
          \end{itemize}
        \end{itemize}
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Kiểm Tra Nhanh Phản Xạ Chân Trị (Phần 1)}
  \begin{cozyExample}[Bài toán: Cho $p = 1$ (Đúng) và $q = 0$ (Sai). Tính chân trị các biểu thức sau:]
    \small
    \begin{columns}[c, onlytextwidth]
      \begin{column}{0.48\textwidth}
        \centering\renewcommand{\arraystretch}{1.15}
        \begin{tabular}{rll}
          \toprule
          \textbf{STT} & \textbf{Biểu thức} & \textbf{Chân trị} \\
          \midrule
          \textbf{1.} & $\neg p = \neg 1$ & $= \mathbf{\textcolor{cozyFalse}{0}}$ \\
          \textbf{2.} & $p \lor q = 1 \lor 0$ & $= \mathbf{\textcolor{cozyTrue}{1}}$ \\
          \textbf{3.} & $p \land q = 1 \land 0$ & $= \mathbf{\textcolor{cozyFalse}{0}}$ \\
          \textbf{4.} & $p \oplus q = 1 \oplus 0$ & $= \mathbf{\textcolor{cozyTrue}{1}}$ \\
          \bottomrule
        \end{tabular}
      \end{column}
      \begin{column}{0.48\textwidth}
        \centering\renewcommand{\arraystretch}{1.15}
        \begin{tabular}{rll}
          \toprule
          \textbf{STT} & \textbf{Biểu thức} & \textbf{Chân trị} \\
          \midrule
          \textbf{5.} & $p \to q = 1 \to 0$ & $= \mathbf{\textcolor{cozyFalse}{0}}$ \\
          \textbf{6.} & $q \to p = 0 \to 1$ & $= \mathbf{\textcolor{cozyTrue}{1}}$ \\
          \textbf{7.} & $p \leftrightarrow q = 1 \leftrightarrow 0$ & $= \mathbf{\textcolor{cozyFalse}{0}}$ \\
          \multicolumn{3}{c}{\footnotesize\color{cozyMuted}Khác chân trị $\implies 0$} \\
          \bottomrule
        \end{tabular}
      \end{column}
    \end{columns}
  \end{cozyExample}
  \vspace{0.05cm}
  \begin{cozyTrap}[Điểm Nhấn Sư Phạm: Vì sao $p \to q \neq q \to p$?]
    \small
    \begin{itemize}
      \setlength{\itemsep}{2pt}
      \item $p \to q = 1 \to 0$: Tiền đề Đúng nhưng Kết luận Sai $\implies$ Thất hứa $\implies$ \textbf{\textcolor{cozyFalse}{Sai ($0$)}}.
      \item $q \to p = 0 \to 1$: Tiền đề Sai ($q=0$) $\implies$ Mệnh đề hiển nhiên \textbf{\textcolor{cozyTrue}{Đúng ($1$)}} (Vacuous Truth).
      \item \textbf{Quy tắc cốt lõi:} Phép kéo theo \textbf{không có tính giao hoán}: $p \to q \not\equiv q \to p$.
    \end{itemize}
  \end{cozyTrap}
\end{frame}

\begin{frame}{Kiểm Tra Nhanh Phản Xạ Chân Trị (Phần 2)}
  \begin{cozyExample}[Xét tính Đúng / Sai của các phát biểu kéo theo ngôn ngữ tự nhiên:]
    \small
    \begin{enumerate}
      \setlength{\itemsep}{4pt}
      \item \textbf{Phát biểu 1:} ``Nếu $0 > 2$ thì Paris là thủ đô của Pháp.''
      \begin{itemize}
        \setlength{\itemsep}{1pt}
        \item \textbf{Phân tích:} Tiền đề $p$: ``$0 > 2$'' (\textbf{\textcolor{cozyFalse}{Sai -- 0}}), Kết luận $q$: ``Paris là thủ đô Pháp'' (\textbf{\textcolor{cozyTrue}{Đúng -- 1}}).
        \item \textbf{Kết luận:} Vì tiền đề Sai $\implies$ Mệnh đề kéo theo \textbf{\textcolor{cozyTrue}{Đúng (1)}}! (Vacuous Truth).
      \end{itemize}

      \item \textbf{Phát biểu 2:} ``Nếu $2 + 2 = 4$ thì $1 + 1 = 3$.''
      \begin{itemize}
        \setlength{\itemsep}{1pt}
        \item \textbf{Phân tích:} Tiền đề $p$: ``$2 + 2 = 4$'' (\textbf{\textcolor{cozyTrue}{Đúng -- 1}}), Kết luận $q$: ``$1 + 1 = 3$'' (\textbf{\textcolor{cozyFalse}{Sai -- 0}}).
        \item \textbf{Kết luận:} Rơi vào trường hợp duy nhất $1 \to 0$ $\implies$ Mệnh đề \textbf{\textcolor{cozyFalse}{Sai (0)}}!
      \end{itemize}

      \item \textbf{Phát biểu 3:} ``Nếu Mặt trời mọc ở hướng Tây thì $1 + 1 = 2$.''
      \begin{itemize}
        \setlength{\itemsep}{1pt}
        \item \textbf{Phân tích:} Tiền đề $p$: ``Mặt trời mọc ở hướng Tây'' (\textbf{\textcolor{cozyFalse}{Sai -- 0}}).
        \item \textbf{Kết luận:} Tiền đề Sai $\implies$ Bất kể kết luận ra sao, mệnh đề hiển nhiên \textbf{\textcolor{cozyTrue}{Đúng (1)}}!
      \end{itemize}
    \end{enumerate}
  \end{cozyExample}
\end{frame}

\begin{frame}{Kỹ Năng Lập Bảng Chân Lý \& Mẹo Nhị Phân (Binary Trick)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{cozyDef}[Quy Tắc $2^n$ Dòng Chân Trị]
        \small
        Một biểu thức chứa $n$ biến mệnh đề độc lập có đúng \textbf{$2^n$ tổ hợp chân trị} (tương ứng $2^n$ dòng trong bảng chân lý):
        \begin{itemize}
          \setlength{\itemsep}{2pt}
          \item \textbf{$1$ biến ($p$):} $2^1 = 2$ dòng ($1, 0$).
          \item \textbf{$2$ biến ($p, q$):} $2^2 = 4$ dòng ($11, 10, 01, 00$).
          \item \textbf{$3$ biến ($p, q, r$):} $2^3 = 8$ dòng.
          \item \textbf{$4$ biến ($p, q, r, s$):} $2^4 = 16$ dòng.
        \end{itemize}
        \vspace{0.05cm}
        \par\footnotesize\color{cozyMuted}Số dòng tăng theo hàm mũ $\implies$ Cần quy luật điền bit chuẩn xác để không bao giờ bị trùng hoặc sót dòng!
      \end{cozyDef}
    \end{column}
    \begin{column}{0.485\textwidth}
      \begin{cozyMethod}[Mẹo Điền Bit Không Bao Giờ Nhầm/Sót Dòng]
        \small
        Để lập bảng cho $n$ biến ($2^n$ dòng, ví dụ 3 biến $p, q, r$ có $8$ dòng):
        \begin{itemize}
          \setlength{\itemsep}{2pt}
          \item \textbf{Cột biến thứ 1 ($p$):} Điền \textbf{nửa trên là 1, nửa dưới là 0}\\
          ($4$ số $1$, tiếp theo là $4$ số $0$).
          \item \textbf{Cột biến thứ 2 ($q$):} Điền xen kẽ từng cụm \textbf{2 số 1, 2 số 0}\\
          ($11, 00, 11, 00$).
          \item \textbf{Cột biến thứ 3 ($r$):} Điền xen kẽ từng \textbf{số 1, 0, 1, 0\dots}\\
          ($1, 0, 1, 0, 1, 0, 1, 0$).
        \end{itemize}
        \vspace{0.05cm}
        \par\footnotesize\color{cozyWarm}\textbf{Bản chất:} Đếm lùi nhị phân từ $(111)_2 = 7$ về $(000)_2 = 0$.
      \end{cozyMethod}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ví Dụ Bảng Chân Lý 2 Biến: Kiểm Tra Tương Đương Logic}
  \begin{cozyExample}[Bài toán: Kiểm tra hai biểu thức $A = \neg(p \lor (\neg p \land q))$ và $B = \neg p \land \neg q$ có tương đương không?]
    \centering
    \renewcommand{\arraystretch}{1.05}
    \footnotesize
    \begin{tabular}{cc|cccc|cc}
      \toprule
      $p$ & $q$ & $\neg p$ & $\neg p \land q$ & $p \lor (\neg p \land q)$ & $A = \neg(p \lor (\neg p \land q))$ & $\neg q$ & $B = \neg p \land \neg q$ \\
      \midrule
      $1$ & $1$ & $0$ & $0$ & $1$ & \textbf{\textcolor{cozyFalse}{0}} & $0$ & \textbf{\textcolor{cozyFalse}{0}} \\
      $1$ & $0$ & $0$ & $0$ & $1$ & \textbf{\textcolor{cozyFalse}{0}} & $1$ & \textbf{\textcolor{cozyFalse}{0}} \\
      $0$ & $1$ & $1$ & $1$ & $1$ & \textbf{\textcolor{cozyFalse}{0}} & $0$ & \textbf{\textcolor{cozyFalse}{0}} \\
      $0$ & $0$ & $1$ & $0$ & $0$ & \textbf{\textcolor{cozyTrue}{1}} & $1$ & \textbf{\textcolor{cozyTrue}{1}} \\
      \bottomrule
    \end{tabular}
  \end{cozyExample}
  \vspace{0.03cm}
  \begin{cozyInsight}[Kết Luận \& Định Nghĩa Tương Đương Logic]
    \small
    \begin{itemize}
      \setlength{\itemsep}{1.5pt}
      \item Hai cột kết quả của \textbf{$A$} và \textbf{$B$} \textbf{giống hệt nhau} trên từng dòng ($0, 0, 0, 1$).
      \item $\implies$ Hai biểu thức \textbf{tương đương logic}: $\neg(p \lor (\neg p \land q)) \equiv \neg p \land \neg q$.
      \item \textbf{Định nghĩa chuẩn mực:} Hai mệnh đề phức hợp $A$ và $B$ tương đương logic ($A \equiv B$ hoặc $A \Leftrightarrow B$) khi và chỉ khi mệnh đề tương đương $A \leftrightarrow B$ là một \textbf{hằng đúng} (nhận giá trị $1$ ở mọi trường hợp).
    \end{itemize}
  \end{cozyInsight}
\end{frame}

\begin{frame}{Bài Tập: Chứng Minh Bằng Bảng Chân Lý (Bài b \& c)}

    \begin{column}{0.44\textwidth}
      \begin{cozyInsight}[Phân Tích 2 Đẳng Thức Tương Đương]
        \footnotesize
        \textbf{1. Câu (b) -- Luật Xuất Khẩu (Exportation):}
        $$p \to (q \to r) \equiv (p \land q) \to r$$

          \item Cột 5 và cột 6 \textbf{trùng khớp hoàn toàn}.
          \item \textit{Ý nghĩa:} ``Nếu có $p$ thì (nếu có $q$ sẽ có $r$)'' $\equiv$ ``Nếu có cả $p$ và $q$ thì có $r$''.
        \end{itemize}
        \vspace{0.06cm}
        \textbf{2. Câu (c) -- Phân Rã Tiền Đề Hội:}
        $$(p \land q) \to r \equiv (p \to r) \lor (q \to r)$$
        \begin{itemize}
          \setlength{\itemsep}{1pt}
          \item Cột 6 và cột 7 \textbf{trùng khớp hoàn toàn}.
          \item Cả 2 vế khẳng định tương đương logic tuyệt đối.
        \end{itemize}

\begin{frame}{Hệ Thống 11 Luật Tương Đương: Cụm 1 -- Tương Đồng Số Học}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{cozyDef}[Bảng Luật Cụm 1: Tương Đồng Số Học]
        \centering
        \renewcommand{\arraystretch}{1.04}
        \small
        \setlength{\tabcolsep}{3pt}
        \begin{tabular}{ll}
          \toprule
          \textbf{Tên Luật} & \textbf{Công Thức Tương Đương} \\
          \midrule
          \textbf{1. Giao hoán} & $p \lor q \equiv q \lor p$ \\
                                & $p \land q \equiv q \land p$ \\
          \textbf{2. Kết hợp}   & $(p \lor q) \lor r \equiv p \lor (q \lor r)$ \\
                                & $(p \land q) \land r \equiv p \land (q \land r)$ \\
          \textbf{3. Đồng nhất} & $p \land \mathbf{T} \equiv p$, \quad $p \lor \mathbf{F} \equiv p$ \\
          \textbf{4. Nuốt}      & $p \lor \mathbf{T} \equiv \mathbf{T}$, \quad $p \land \mathbf{F} \equiv \mathbf{F}$ \\
          \textbf{5. Phủ định kép} & $\neg(\neg p) \equiv p$ \\
          \bottomrule
        \end{tabular}
      \end{cozyDef}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{cozyInsight}[Trực Giác Số Học Đối Chiếu]
        \small
        \begin{itemize}
          \setlength{\itemsep}{1pt}
          \item \textbf{Phép $\lor \sim$ Phép cộng ($+$):}\\
          $\mathbf{F}$ ($0$) là phần tử trung hòa ($x + 0 = x \longleftrightarrow p \lor \mathbf{F} \equiv p$).
          \item \textbf{Phép $\land \sim$ Phép nhân ($\times$):}\\
          $\mathbf{T}$ ($1$) là phần tử đơn vị ($x \times 1 = x \longleftrightarrow p \land \mathbf{T} \equiv p$).
          \item \textbf{Tính chất nuốt (Domination):}\\
          Số $0$ nuốt trong phép nhân ($x \times 0 = 0 \longleftrightarrow p \land \mathbf{F} \equiv \mathbf{F}$); $\mathbf{T}$ nuốt trong phép tuyển ($p \lor \mathbf{T} \equiv \mathbf{T}$).
          \item \textbf{Phủ định kép:}\\
          Hai lần đổi dấu là khẳng định ($-(-x) = x$).
        \end{itemize}
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Hệ Thống 11 Luật Tương Đương: Cụm 2 -- Đặc Thù Đại Số Boole}
  \begin{columns}[t, onlytextwidth]

      \begin{cozyDef}[Bảng Luật Cụm 2: Đặc Thù Đại Số Boole]
        \centering
        \renewcommand{\arraystretch}{1.08}
        \small
        \begin{tabular}{ll}
          \toprule
          \textbf{Tên Luật} & \textbf{Công Thức Tương Đương} \\
          \midrule
          \textbf{6. Lũy đẳng}   & $p \lor p \equiv p$ \\
                                 & $p \land p \equiv p$ \\
          \textbf{7. Phân phối}  & $p \lor (q \land r) \equiv (p \lor q) \land (p \lor r)$ \\
                                 & $p \land (q \lor r) \equiv (p \land q) \lor (p \land r)$ \\
          \textbf{8. Hấp thụ}    & $p \lor (p \land q) \equiv p$ \\
                                 & $p \land (p \lor q) \equiv p$ \\
          \textbf{9. Phần bù}    & $p \lor \neg p \equiv \mathbf{T}$ \\
                                 & $p \land \neg p \equiv \mathbf{F}$ \\
          \bottomrule
        \end{tabular}
      \end{cozyDef}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{cozyTrap}[Cạm Bẫy Phân Phối \& Mẹo Hấp Thụ]
        \small
        \begin{itemize}
          \setlength{\itemsep}{2pt}
          \item \textbf{Phân phối 2 chiều:}\\
          Trong số học chỉ có $a(b+c) = ab+ac$, \textbf{không có} $a+(bc)=(a+b)(a+c)$. Trong logic Boole, phân phối \textbf{đối xứng cả hai chiều} ($\lor$ cho $\land$ và $\land$ cho $\lor$)!
          \item \textbf{Mẹo nhớ Luật Hấp thụ:}\\
          Biến $p$ đứng ngoài và lặp lại bên trong ngoặc với toán tử ngược: $p \lor (p \land q) \equiv p$ $\implies p$ ``hấp thụ'' toàn bộ ngoặc!
        \end{itemize}
      \end{cozyTrap}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Hệ Thống 11 Luật Tương Đương: Cụm 3 -- Chuyển Đổi Toán Tử}

          $$\lor \longleftrightarrow \land \quad \text{và} \quad \mathbf{T} \longleftrightarrow \mathbf{F}$$
          ta lập tức thu được một đẳng thức logic \textbf{luôn đúng mới}!
          \item \textbf{Ví dụ đối ngẫu:}\\
          {\footnotesize $\bullet$ $p \lor (p \land q) \equiv p \;\longleftrightarrow\; p \land (p \lor q) \equiv p$}\\
          {\footnotesize $\bullet$ $p \lor \neg p \equiv \mathbf{T} \;\longleftrightarrow\; p \land \neg p \equiv \mathbf{F}$}
          \item $\implies$ Giảm \textbf{$50\%$ công sức} học~thuộc~lòng!
        \end{itemize}

\begin{frame}{Thế Nào Là Một Lập Luận Hợp Lệ? (Valid Argument)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{cozyDef}[Định Nghĩa Lập Luận Hợp Lệ]
        \small
        \begin{itemize}
          \setlength{\itemsep}{1.5pt}
          \item Một \textbf{lập luận} gồm các \textbf{tiền đề} $P_1, P_2, \dots, P_k$ và \textbf{kết luận} $C$, ký hiệu:
          $$P_1, P_2, \dots, P_k \vdash C \quad (\therefore C)$$
          \item Lập luận là \textbf{\textcolor{cozyTrue}{HỢP LỆ (Valid)}} khi và chỉ khi:
          $$(P_1 \land P_2 \land \dots \land P_k) \to C$$
          là một \textbf{HẰNG ĐÚNG (Tautology)}.
          \item \textbf{Quy tắc vàng:} Nếu \textbf{TẤT CẢ} các tiền đề đều Đúng, thì kết luận \textbf{BẮT BUỘC} phải Đúng! (Không bao giờ xảy ra $1 \to 0$).
        \end{itemize}
      \end{cozyDef}
    \end{column}
    \begin{column}{0.485\textwidth}
      \begin{cozyInsight}[Hợp Lệ (Valid) vs Chân Thực (Sound)]
        \small
        \begin{itemize}
          \setlength{\itemsep}{1.5pt}
          \item \textbf{Hợp lệ (Validity):} Chỉ quan tâm đến \textbf{cấu trúc hình thức}, không phụ thuộc thực tế.
          \begin{itemize}
            \setlength{\itemsep}{0.8pt}
            \item \textit{Ví dụ:} ``Mọi loài chim đều biết bay. Chim cánh cụt là chim $\implies$ Chim cánh cụt biết bay.''
            \item Cấu trúc lập luận này \textbf{\textcolor{cozyTrue}{HỢP LỆ}}, dù tiền đề $1$ sai trong sinh học!
          \end{itemize}
          \item \textbf{Chân thực (Soundness):} Lập luận vừa \textbf{Hợp lệ} vừa có \textbf{toàn bộ tiền đề Đúng trong thực tế}.
          \item \textbf{Trọng tâm môn học:} Toán học \& Tin học tập trung kiểm định \textbf{tính hợp lệ hình thức}.
        \end{itemize}
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{8 Quy Tắc Suy Luận Nền Tảng (Phần 1: Kéo Theo \& Tuyển)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{cozyDef}[1. Modus Ponens (Khẳng định tiền đề)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} p \\ p \to q \\ \hline \therefore q \end{array} \quad \Big( [p \land (p \to q)] \to q \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Tiền đề $p$ đúng $\implies$ Kéo theo $q$ xảy ra.\\
        \textit{Ví dụ:} Hôm nay mưa. Nếu mưa thì đường ướt $\implies$ Đường ướt.
      \end{cozyDef}
      \vspace{0.08cm}
      \begin{cozyDef}[3. Tam đoạn luận giả định (Bắc cầu)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} p \to q \\ q \to r \\ \hline \therefore p \to r \end{array} \quad \Big( [(p \to q) \land (q \to r)] \to (p \to r) \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Tính bắc cầu qua mắt xích trung gian $q$.\\
        \textit{Ví dụ:} Học chăm $\to$ hiểu bài; hiểu bài $\to$ qua môn $\implies$ Học chăm $\to$ qua môn.
      \end{cozyDef}
    \end{column}
    \begin{column}{0.485\textwidth}
      \begin{cozyDef}[2. Modus Tollens (Phủ định kết luận)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} \neg q \\ p \to q \\ \hline \therefore \neg p \end{array} \quad \Big( [\neg q \land (p \to q)] \to \neg p \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Kết luận sai $\implies$ Tiền đề bắt buộc~sai.\\
        \textit{Ví dụ:} Đường không ướt. Nếu mưa thì đường ướt $\implies$ Trời không mưa.
      \end{cozyDef}
      \vspace{0.08cm}
      \begin{cozyDef}[4. Tam đoạn luận tuyển (Loại trừ)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} p \lor q \\ \neg p \\ \hline \therefore q \end{array} \quad \Big( [(p \lor q) \land \neg p] \to q \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Một trong hai phải đúng, loại cái này $\implies$ Chọn cái kia.\\
        \textit{Ví dụ:} Đi xe buýt hoặc đi bộ. Không đi xe buýt $\implies$ Đi bộ.
      \end{cozyDef}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{8 Quy Tắc Suy Luận Nền Tảng (Phần 2: Thao Tác \& Phân Giải)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{cozyDef}[5. Quy Tắc Cộng (Addition)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} p \\ \hline \therefore p \lor q \end{array} \quad \Big( p \to (p \lor q) \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Khi $p$ đã đúng, tuyển với bất kỳ $q$ nào cũng đúng.\\
        \textit{Ví dụ:} $3$ là số nguyên tố $\implies 3$ là số nguyên tố hoặc $3$ là số chẵn.
      \end{cozyDef}
      \vspace{0.08cm}
      \begin{cozyDef}[7. Quy Tắc Ghép Hội (Conjunction)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} p \\ q \\ \hline \therefore p \land q \end{array} \quad \Big( (p \land q) \to (p \land q) \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Hai sự thật độc lập cùng đúng thì phép hội đúng.\\
        \textit{Ví dụ:} Bạn học chăm ($p$). Bạn thông minh ($q$) $\implies$ Bạn vừa chăm vừa thông minh ($p \land q$).
      \end{cozyDef}
    \end{column}
    \begin{column}{0.485\textwidth}
      \begin{cozyDef}[6. Quy Tắc Rút Gọn (Simplification)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} p \land q \\ \hline \therefore p \end{array} \quad \Big( (p \land q) \to p \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Cả hai cùng đúng thì hiển nhiên từng cái đúng.\\
        \textit{Ví dụ:} Trời mưa và lạnh $\implies$ Trời~mưa.
      \end{cozyDef}
      \vspace{0.08cm}
      \begin{cozyInsight}[8. Quy Tắc Phân Giải (Resolution)]
        \scriptsize
        \vspace{-0.1cm}
        $$\begin{array}{l} p \lor q \\ \neg p \lor r \\ \hline \therefore q \lor r \end{array} \quad \Big( [(p \lor q) \land (\neg p \lor r)] \to (q \lor r) \equiv \mathbf{T} \Big)$$
        \vspace{-0.1cm}
        \textbf{Ý nghĩa:} Triệt tiêu cặp bù nhau $(p, \neg p)$ giữa hai tuyển.\\
        \textbf{Ứng dụng:} Nền tảng của các hệ chứng minh tự động trong Trí tuệ nhân tạo (AI) và Lập trình Prolog!
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cảnh Báo: 2 Cạm Bẫy Ngụy Biện Bài Thi Thường Gặp (Formal Fallacies)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{cozyTrap}[2 Ngụy Biện Kinh Điển Phải Tuyệt Đối Tránh]
        \small
        \begin{enumerate}
          \setlength{\itemsep}{2.5pt}
          \item \textbf{Khẳng định kết luận (Affirming Consequent):}\\
          Có $p \to q$ và $q \;\not\vdash\; p$ \textbf{\textcolor{cozyFalse}{(SAI!)}}.\\
          \textit{Ví dụ:} ``Nếu trời mưa thì đường ướt. Đường ướt $\to$ Trời mưa.''\\
          $\implies$ \textbf{Sai}, xe bồn tưới nước cũng làm đường ướt!
          \item \textbf{Phủ định tiền đề (Denying Antecedent):}\\
          Có $p \to q$ và $\neg p \;\not\vdash\; \neg q$ \textbf{\textcolor{cozyFalse}{(SAI!)}}.\\
          \textit{Ví dụ:} ``Nếu là cá heo thì biết bơi. Tôi không phải cá heo $\to$ Tôi không biết bơi.''\\
          $\implies$ \textbf{Sai}, người và vịt vẫn bơi rất giỏi!
        \end{enumerate}
      \end{cozyTrap}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{cozyInsight}[Bản Chất Góc Nhìn Chân Trị]
        \small
        \begin{itemize}
          \setlength{\itemsep}{2pt}
          \item Phép kéo theo $p \to q$ chỉ cho phép suy luận 2 hướng chuẩn:
          \begin{itemize}
            \item \textbf{Thuận:} $p \implies q$ (Modus Ponens)
            \item \textbf{Phản đảo:} $\neg q \implies \neg p$ (Modus Tollens)
          \end{itemize}
          \item Khi $p = 0$, $p \to q$ luôn \textbf{tự động nhận giá trị Đúng} (Vacuous Truth) dù $q = 1$ hay $q = 0$.
          \item $\implies$ Biết $q$ Đúng hoặc $\neg p$ Đúng \textbf{hoàn toàn không cho phép suy ra} biến còn lại!
        \end{itemize}
      \end{cozyInsight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Giải Mã Loạt Bài Tập Suy Luận Ngôn Ngữ Tự Nhiên (Câu d -- i)}
  \begin{cozyExample}[Phân Tích Tính Hợp Lệ Của 6 Câu Hỏi Ngôn Ngữ Tự Nhiên]
    \footnotesize
    \begin{columns}[t, onlytextwidth]
      \begin{column}{0.485\textwidth}
        \begin{itemize}
          \setlength{\itemsep}{2pt}
          \item \textbf{(d) ``Mưa $\to$ ở nhà. Ở nhà $\to$ Mưa'':}\\
          Cấu trúc: $p \to q, q \vdash p \implies$ \textbf{\textcolor{cozyFalse}{Khẳng định kết luận (Sai)}}.
          \item \textbf{(e) ``Chăm chỉ $\to$ qua môn. Qua môn $\to$ Chăm'':}\\
          Cấu trúc: $p \to q, q \vdash p \implies$ \textbf{\textcolor{cozyFalse}{Khẳng định kết luận (Sai)}}.
          \item \textbf{(f) ``$x > 2 \to x^2 > 4$. Cho $x \le 2 \to x^2 \le 4$'':}\\
          Cấu trúc: $p \to q, \neg p \vdash \neg q \implies$ \textbf{\textcolor{cozyFalse}{Phủ định tiền đề (Sai)}}.\\
          \scriptsize\textcolor{cozyMuted}{(Phản ví dụ: $x = -3 \le 2$ nhưng $(-3)^2 = 9 > 4$!)}

      \end{column}
      \begin{column}{0.485\textwidth}

          \setlength{\itemsep}{2pt}
          \item \textbf{(g) ``Không ăn sáng $\to$ đói. Ăn sáng $\to$ không đói'':}\\
          Cấu trúc: $\neg p \to q, p \vdash \neg q \implies$ \textbf{\textcolor{cozyFalse}{Phủ định tiền đề (Sai)}}.
          \item \textbf{(h) ``Yêu $\to$ tặng quà. Không yêu $\to$ không tặng'':}\\
          Cấu trúc: $p \to q, \neg p \vdash \neg q \implies$ \textbf{\textcolor{cozyFalse}{Phủ định tiền đề (Sai)}}.
          \item \textbf{(i) ``Lửa $\to$ khói. Có khói $\to$ Có lửa'':}\\
          Cấu trúc: $p \to q, q \vdash p \implies$ \textbf{\textcolor{cozyFalse}{Khẳng định kết luận (Sai)}}.\\
          \scriptsize\textcolor{cozyMuted}{(Khói có thể do đá khô, máy tạo khói sân khấu).}
        \end{itemize}
      \end{column}
    \end{columns}
  \end{cozyExample}
  \vspace{0.03cm}
  \par\centering\footnotesize\color{cozyWarm}\textbf{Quy tắc vàng bài thi:} Kéo theo 1 chiều ($p \to q$) không tương đương với 2 chiều ($p \leftrightarrow q$)!
\end{frame}

\begin{frame}{Lời Giải Chi Tiết: Kiểm Tra Lập Luận Hình Thức (Bài a)}
  \begin{columns}[t, onlytextwidth]

      \begin{cozyExample}[Bài (a): $p, \; q \to r, \; p \to \neg r \;\vdash\; \neg q$]

        \renewcommand{\arraystretch}{1.1}
        \footnotesize
        \setlength{\tabcolsep}{3pt}
        \begin{tabular}{cll}
          \toprule
          \textbf{Bước} & \textbf{Mệnh đề} & \textbf{Lý do / Quy tắc áp dụng} \\
          \midrule
          1. & $p$ & Tiền đề 1 \\
          2. & $p \to \neg r$ & Tiền đề 2 \\
          3. & $\neg r$ & \textbf{Modus Ponens} từ (1) và (2) \\
          4. & $q \to r$ & Tiền đề 3 \\
          5. & $\mathbf{\textcolor{cozyTrue}{\neg q}}$ & \textbf{Modus Tollens} từ (3) và (4) \\
          \bottomrule
        \end{tabular}
        \par\vspace{0.08cm}
        \centering\footnotesize\color{cozyTrue}$\therefore$ Kết luận $\neg q$ được suy ra $\implies$ Lập luận \textbf{HỢP LỆ}!
      \end{cozyExample}
    \end{column}
    \begin{column}{0.44\textwidth}

        \small
        \begin{itemize}
          \setlength{\itemsep}{2.5pt}
          \item \textbf{Khởi đầu từ biến đơn:}\\
          Bắt đầu từ mệnh đề đơn $p$ đã biết chân trị Đúng.
          \item \textbf{Mở khóa qua Modus Ponens:}\\

          \item \textbf{Về đích qua Modus Tollens:}\\
          Có $\neg r$ (phủ định kết luận của $q \to r$) $\implies$ Buộc tiền đề $q$ phải Sai ($\neg q$).
        \end{itemize}

\begin{frame}{Chinh Phục Bài Toán Thực Tế: Quy Trình 4 Bước Hình Thức Hóa}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{cozyMethod}[Quy Trình 4 Bước Giải Bài Toán Suy Luận Thực Tế]
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.8pt}
          \item \textbf{Bước 1 (Nguyên tử hóa biến):}\\
          Đặt các biến mệnh đề đơn $p, q, r, s, t$ cho các câu khẳng định cơ sở (luôn đặt ở thể khẳng định).
          \item \textbf{Bước 2 (Hình thức hóa giả thiết):}\\
          Dịch từng câu văn trong đề bài thành các tiền đề logic $P_1, P_2, \dots, P_k$ và kết luận $C$.
          \item \textbf{Bước 3 (Lập chiến lược suy diễn):}\\
          Xác định điểm khởi đầu (mệnh đề hội hoặc chân trị đơn), dùng Modus Ponens / Tollens mở khóa từng bước.
          \item \textbf{Bước 4 (Dựng bảng chứng minh):}\\
          Lập bảng 3 cột chuẩn mực, đối chiếu từng dòng với tên quy tắc cụ thể.
        \end{itemize}
      \end{cozyMethod}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{cozyTrap}[Từ Khóa Dễ Bẫy Khi Dịch Đề]
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.8pt}
          \item \textbf{``$A$ chỉ khi $B$'':} $\implies A \to B$\\
          {\footnotesize\color{cozyMuted}(Không phải $B \to A$! $B$ là điều kiện cần).}
          \item \textbf{``$A$ khi $B$'':} $\implies B \to A$\\
          {\footnotesize\color{cozyMuted}($B$ là điều kiện đủ để có $A$)}.
          \item \textbf{``Trừ khi $A$, thì $B$'':} $\implies \neg A \to B$.
          \item \textbf{``$A$ nhưng $B$'':} $\implies A \land B$\\
          {\footnotesize\color{cozyMuted}(Bản chất là phép hội đồng thời)}.
        \end{itemize}
      \end{cozyTrap}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Bài Toán Dã Ngoại: Đi Bơi Hay Đi Canô? (Đề Bài \& Đặt Biến)}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{cozyDef}[Đề Bài Ngôn Ngữ Tự Nhiên]
        \small
        Cho các sự kiện trong chuyến dã ngoại:
        \begin{enumerate}
          \setlength{\itemsep}{0.5pt}
          \item ``Chiều nay trời không nắng và trời lạnh hơn hôm qua.''
          \item ``Chúng ta sẽ đi bơi \textbf{chỉ khi} trời nắng.''
          \item ``Nếu chúng ta không đi bơi, chúng ta sẽ đi tham quan bằng canô.''
          \item ``Nếu đi tham quan bằng canô, chúng ta sẽ về nhà trước khi mặt trời lặn.''
        \end{enumerate}
        \par\vspace{0.02cm}
        \textbf{Yêu cầu:} Chứng minh kết luận: \textit{``Chúng ta sẽ về nhà trước khi mặt trời lặn''} là hợp lệ.
      \end{cozyDef}
    \end{column}
    \begin{column}{0.485\textwidth}
      \begin{cozyMethod}[Bước 1 \& 2: Đặt Biến \& Hình Thức Hóa]
        \footnotesize
        \textbf{1. Đặt 5 biến mệnh đề sơ cấp:}
        \begin{itemize}
          \setlength{\itemsep}{0.3pt}
          \item $p$: ``Chiều nay trời nắng.''
          \item $q$: ``Trời lạnh hơn hôm qua.''
          \item $r$: ``Chúng ta đi bơi.''
          \item $s$: ``Chúng ta đi tham quan bằng canô.''
          \item $t$: ``Chúng ta về nhà trước khi mặt trời lặn.''
        \end{itemize}
        \textbf{2. Dịch thành 4 tiền đề \& kết luận:}
        \begin{itemize}
          \setlength{\itemsep}{0.3pt}
          \item Tiền đề 1: $\neg p \land q$
          \item Tiền đề 2: $r \to p$ (\textit{``$r$ chỉ khi $p$''})
          \item Tiền đề 3: $\neg r \to s$
          \item Tiền đề 4: $s \to t$
          \item \textbf{Cần chứng minh:} $t$
        \end{itemize}
      \end{cozyMethod}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Bài Toán Dã Ngoại: Bảng Suy Luận 8 Bước Hoàn Chỉnh}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.56\textwidth}
      \begin{cozyExample}[Bảng Suy Luận Logic Hình Thức 8 Bước]
        \centering
        \renewcommand{\arraystretch}{1.03}
        \footnotesize
        \setlength{\tabcolsep}{2.5pt}
        \begin{tabular}{cll}
          \toprule
          \textbf{Bước} & \textbf{Mệnh đề} & \textbf{Lý do / Quy tắc áp dụng} \\
          \midrule
          1. & $\neg p \land q$ & Tiền đề 1 \\
          2. & $\neg p$ & Quy tắc Rút gọn từ (1) \\
          3. & $r \to p$ & Tiền đề 2 \\
          4. & $\mathbf{\textcolor{cozyFalse}{\neg r}}$ & \textbf{Modus Tollens} từ (2) và (3) \\
          5. & $\neg r \to s$ & Tiền đề 3 \\
          6. & $\mathbf{\textcolor{cozyWarm}{s}}$ & \textbf{Modus Ponens} từ (4) và (5) \\

          8. & $\mathbf{\textcolor{cozyTrue}{t}}$ & \textbf{Modus Ponens} từ (6) và (7) \\

        \centering\footnotesize\color{cozyTrue}$\therefore$ Kết luận được chứng minh hợp lệ tuyệt đối!
      \end{cozyExample}

          \item Biến mệnh đề sơ cấp ($p, q$) $\to$ Phức hợp.

        \end{itemize}
      \end{cozyDef}

          \item Tương đương ($p \leftrightarrow q$): Cùng chân trị thì đúng.
